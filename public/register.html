curl -H "Authorization: Bearer YOUR_KEY" \ "https://senlay.cloud/api/v1/sense?lat=36.01&lng=-5.60"
import requests API_KEY = "YOUR_KEY" headers = {"Authorization": f"Bearer {API_KEY}"} # Plain text — easiest to inject into LLM context r = requests.get( "https://senlay.cloud/api/v1/sense", headers=headers, params={"lat": 36.01, "lng": -5.60} ) print(r.text) # Or structured JSON with the source data available for this request r = requests.get( "https://senlay.cloud/api/v1/pwm", headers=headers, params={"lat": 36.01, "lng": -5.60} ) data = r.json() print(data["context_string"])
const API_KEY = "YOUR_KEY"; // Plain text — drop straight into your prompt const res = await fetch( "https://senlay.cloud/api/v1/sense?lat=36.01&lng=-5.60", { headers: { Authorization: `Bearer ${API_KEY}` } } ); const sensoryContext = await res.text(); console.log(sensoryContext); // Or full JSON const json = await fetch( "https://senlay.cloud/api/v1/pwm?lat=36.01&lng=-5.60", { headers: { Authorization: `Bearer ${API_KEY}` } } ).then(r => r.json()); console.log(json.context_string);
import anthropic, requests senlay = requests.get( "https://senlay.cloud/api/v1/sense", headers={"Authorization": "Bearer YOUR_KEY"}, params={"lat": 36.01, "lng": -5.60} ).text client = anthropic.Anthropic() msg = client.messages.create( model="claude-sonnet-4-20250514", max_tokens=1024, messages=[{ "role": "user", "content": f"{senlay}\n\nGiven these conditions, " f"is it safe to kitesurf right now?" }] ) print(msg.content[0].text)
